            <class name="tests.MoreUIValidationsTest"/>
            <class name="tests.UIValidationContinueTest"/>

<!DOCTYPE suite SYSTEM "https://testng.org/testng-1.0.dtd">
<suite name="TestSuites" verbose="1" parallel="classes" thread-count="2">

    <test name="RegressionTests">

//RUn your maven project with Maven Build, It is necessary for CICD pipeline.. we can find this in "maven testNG integration documents" from https://maven.apache.org/surefire/maven-surefire-plugin/examples/testng.html
// we also need "TestNG Engine For The JUnit Platform" for this in side dependency

// add below code in pom.xml after <depencencies/>, if we want to run based on group
//NOTE: whatever we add or modify, we have to do it within <configuration> only and if we do not provide any configuration tag, then it will run all the test cases from all classes by default.


<plugins>
      <plugin>
        <groupId>org.apache.maven.plugins</groupId>
        <artifactId>maven-surefire-plugin</artifactId>
        <version>3.6.0-M1</version>
        <configuration>
          <groups>smoke</groups>
        </configuration>
      </plugin>
</plugins>


//------------------// do we need to change pom.xml every time if we want run only smoke test or wanted to run all test cases ??? NO.------//----
// In maven there is concept of Profiles same like dependencies
// inside <profiles>, we can add multiple <profile> and in those profile we can define whether to run all test on only smoke test etc...

// Syntax would be ,
        <profiles>
                <profile>
                            <!--profile one for all test cases to run-->
                </profile>

                <profile>
                            <!--profile two for smoke to run-->
                </profile>

            </profiles>

// we separate profile by id, and we place that <build> of surefire plugin inside <profile>
// to run above use command "mvn test -PRegression"      here -P stand for ProfileToRun
//complete syntax

<profiles>
        <profile>
            <!--profile one for all test cases to run-->
            <id>Regression</id>
            <build>
                <plugins>
                    <plugin>
                        <groupId>org.apache.maven.plugins</groupId>
                        <artifactId>maven-surefire-plugin</artifactId>
                        <version>3.6.0-M1</version>
                    </plugin>
                </plugins>

            </build>
        </profile>

        <profile>
            <!--profile two for smoke to run-->
            <id>Smoke</id>
            <build>
                <plugins>
                    <plugin>
                        <groupId>org.apache.maven.plugins</groupId>
                        <artifactId>maven-surefire-plugin</artifactId>
                        <version>3.6.0-M1</version>
                        <configuration>
                            <groups>smoke</groups>
                        </configuration>
                    </plugin>
                </plugins>

            </build>
        </profile>

    </profiles>
